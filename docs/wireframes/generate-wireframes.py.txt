def nav(role, active):
    if role == 'admin':
        links = [('Dashboard','dash'),('Kelola Course','kelola'),('Courses','courses')]
    else:
        links = [('Dashboard','dash'),('Courses','courses')]
    ls = ''.join(f"<span class='link {'active' if k==active else ''}'>{t}</span>" for t,k in links)
    return f"<div class='nav'><div class='logo'>LOGO</div>{ls}<span class='user'>Nama Pengguna ▾</span></div>"
